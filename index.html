Koneksi Database Cloud (Google Sheets)

Hubungkan HP seluruh murid & guru secara gratis 100% selamanya.

Kenapa akun murid belum masuk ke Mode Guru?
Karena murid membuka web dari HP mereka masing-masing sehingga data tersimpan di memori HP mereka. Agar data dari semua HP murid otomatis terkumpul di Mode Guru ini, Anda cukup menyambungkan Google Sheet gratis.

1 Buat Spreadsheet Baru & Buka Apps Script:

Buka sheets.new di browser Google Anda, lalu klik menu Ekstensi (Extensions) > Apps Script.

2 Salin & Tempel Kode Script Lengkap Ini:

function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheetSiswa = ss.getSheetByName("Daftar_Siswa"); var students = []; if (sheetSiswa && sheetSiswa.getLastRow() > 1) { var sData = sheetSiswa.getRange(2, 1, sheetSiswa.getLastRow() - 1, 5).getValues(); students = sData.map(function(r) { return { timestamp: r[0], name: r[1], gradeName: r[2], school: r[3], lastLogin: r[4] }; }); } var sheetNilai = ss.getSheetByName("Rekap_Nilai"); var submissions = []; if (sheetNilai && sheetNilai.getLastRow() > 1) { var nData = sheetNilai.getRange(2, 1, sheetNilai.getLastRow() - 1, 8).getValues(); submissions = nData.map(function(r) { return { timestamp: r[0], studentName: r[1], studentClass: r[2], displaySubject: r[3], correctCount: r[4], totalQuestions: r[5], score: r[6], passed: r[7] === "TUNTAS" || r[7] === true }; }); } return ContentService.createTextOutput(JSON.stringify({ status: "success", students: students, submissions: submissions })).setMimeType(ContentService.MimeType.JSON); } function doPost(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var data = JSON.parse(e.postData.contents); if (data.action === "register_student") { var sSiswa = ss.getSheetByName("Daftar_Siswa") || ss.insertSheet("Daftar_Siswa"); if (sSiswa.getLastRow() === 0) sSiswa.appendRow(["Waktu Terdaftar", "Nama Siswa", "Kelas", "Asal Sekolah", "Aktivitas Terakhir"]); var rows = sSiswa.getDataRange().getValues(), found = -1; for (var i = 1; i < rows.length; i++) { if (rows[i][1].toString().toLowerCase() === data.name.toString().toLowerCase()) { found = i + 1; break; } } var now = data.timestamp || new Date().toLocaleString("id-ID"); if (found > 0) { sSiswa.getRange(found, 3).setValue(data.gradeName || data.grade); sSiswa.getRange(found, 4).setValue(data.school || "Sekolah"); sSiswa.getRange(found, 5).setValue(now); } else { sSiswa.appendRow([now, data.name, data.gradeName || data.grade, data.school || "Sekolah", now]); } return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON); } if (data.action === "submit_exam" || data.score !== undefined) { var sNilai = ss.getSheetByName("Rekap_Nilai") || ss.insertSheet("Rekap_Nilai"); if (sNilai.getLastRow() === 0) sNilai.appendRow(["Waktu Ujian", "Nama Siswa", "Kelas", "Materi Bab", "Benar", "Total Soal", "Skor", "Status"]); sNilai.appendRow([data.timestamp || new Date().toLocaleString("id-ID"), data.studentName, data.studentClass, data.displaySubject || data.chapterTitle || "-", data.correctCount, data.totalQuestions, data.score, data.passed ? "TUNTAS" : "REMEDIAL"]); return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({ status: "unknown" })).setMimeType(ContentService.MimeType.JSON); }
